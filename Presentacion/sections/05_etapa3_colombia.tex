\begin{frame}{Etapa 3: Validación de la señal digital (P1)}
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      \textbf{Mann-Whitney + permutación intra-contienda} \corr{[Corr.\ 1.16/2.2]}
      \begin{itemize}\scriptsize
        \item $p = 0{,}0481$ para \textbf{engagement por publicación}
              (no para diferencia de medias 44.252 vs.\ 28.011)
        \item Candidatos anidados en 31 contiendas; se barajaron etiquetas
              ganador/perdedor \textbf{dentro} de cada ciudad (10.000 iter.)
        \item $\Delta\tilde{x} = 112$ \textit{eng}/publicación observado;
              ninguna permutación lo igualó: \kw{$p < 0{,}0001$ empírico}
      \end{itemize}
      \vspace{0.15em}
      \textbf{Correlación de Spearman} \corr{[Corr.\ 2.2]}
      \begin{itemize}\scriptsize
        \item $\rho = 0{,}70$ dominancia relativa; $\rho \approx 0{,}45$ volumen absoluto
        \item Composicionalidad cuantificada: $\rho_{\text{mec.}} = 0{,}261$ (39\%),
              señal real $\rho_{\text{real}} \approx 0{,}41$ (61\%), $p < 0{,}0001$
      \end{itemize}
      \vspace{0.15em}
      \textbf{Heurístico de dominancia:}
      \[\textstyle
        \Pr(\text{Top-1}) = \tfrac{22}{31} = 71{,}0\%
        \;\;\text{IC}_{95\%}\text{Wilson:}\;[53{,}4\%,\;83{,}9\%]
      \]
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{exampleblock}{\textbf{P1}: Respondida}
        \small La señal existe. El 61\% del $\rho$ es
        señal real ($\rho_{\text{real}} \approx 0{,}41$);
        el 39\% es inflación composicional. La validez
        predictiva definitiva: LOCO-CV, MAE = 9{,}56 pp.
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Etapa 3: Normalización y construcción de variables}
  \textbf{Pool inicial: más de 22 variables}
  (interacciones reconstruidas con el restaurador Weibull)
  \vspace{0.3em}
  \begin{columns}[T]
    \begin{column}{0.30\textwidth}
      \begin{block}{\small 1. Nominales absolutas}
        \scriptsize Likes, comentarios, compartidos totales por plataforma.\\[0.3em]
        \alrt{Problema:} quien más publica, más acumula.
        Señal ruidosa.
      \end{block}
    \end{column}
    \begin{column}{0.30\textwidth}
      \begin{block}{\small 2. Por publicación}
        \scriptsize Likes, comentarios y compartidos \textbf{promedio por post}.\\[0.3em]
        Normalizan la actividad editorial.
        Más estables, pero dependen del tamaño de la contienda.
      \end{block}
    \end{column}
    \begin{column}{0.34\textwidth}
      \begin{exampleblock}{\small 3. Dominancia relativa \checkmark}
        \scriptsize \textbf{Cuota de likes} sobre el total de la contienda.\\[0.3em]
        Elimina el efecto de tamaño absoluto.
        Clasificador naive: \textbf{71\%}; $\rho = 0{,}70$.
      \end{exampleblock}
    \end{column}
  \end{columns}
  \vspace{0.4em}
  \begin{alertblock}{Intento descartado: normalizar por número de seguidores}
    \small VIF elevado. El número de seguidores es inestable, colineal con el volumen
    de publicaciones, y no disponible históricamente en todas las plataformas.
  \end{alertblock}
\end{frame}
