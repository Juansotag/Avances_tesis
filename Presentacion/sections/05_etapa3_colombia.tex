\begin{frame}{Modelado predictivo: regresión logística fraccional}
\begin{columns}[c]
\column{0.52\textwidth}
  \textbf{\color{elanGold}Especificación del modelo}\\[0.4em]
  La variable objetivo es la \textbf{cuota de voto real} $v_c \in (0,1)$.\\[0.5em]
  Dado que $v_c$ es una proporción, usamos regresión logística fraccional (Papke \& Wooldridge, 1996):
  \[
    \mathbb{E}[v_c \mid X] = \frac{1}{1 + e^{-(\beta_0 + X^\top \beta)}}
  \]
  con predictor central:
  \[
    \text{logit}(\delta_c) = \log\!\left(\frac{\delta_c}{1-\delta_c}\right)
  \]
  \vspace{0.3em}
  \textbf{\color{elanGold}Selección de variables: ElasticNet anidado}\\
  \[
    \hat{\beta} = \arg\min_\beta \bigl\{L(\beta) + \lambda_1\|\beta\|_1 + \lambda_2\|\beta\|_2^2\bigr\}
  \]

\column{0.44\textwidth}
  \textbf{\color{elanGold}Variables del modelo final}\\[0.4em]
  \begin{tabular}{lc}
    \toprule
    \color{elanWhite}Variable & \color{elanGold}$\hat{\beta}$\\
    \midrule
    logit(dominancia\_likes) & $0.431$\\
    $\log(\text{población})$ & $-0.012$\\
    \textbf{interacción} & $\mathbf{0.557}$\\
    \bottomrule
  \end{tabular}
  \vspace{0.7em}
  {\color{elanGray}\small El coeficiente más fuerte del modelo es la \textbf{interacción} entre dominancia digital y tamaño de ciudad: en municipios grandes la señal digital se amplifica.}

  \vspace{0.6em}
  \textbf{\color{elanGold}Validación}: LOCO-CV\\
  {\small Leave-One-Contest-Out: métricas honestas fuera de muestra, contienda completa como unidad de test.}
\end{columns}
\end{frame}
