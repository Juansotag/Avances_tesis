\begin{frame}{Variable objetivo: dominancia digital}
\begin{columns}[c]
\column{0.5\textwidth}
  \textbf{\color{elanGold}Definición formal}\\[0.6em]
  Sea $\mathcal{C}_d$ el conjunto de candidatos en la contienda $d$.\\[0.4em]
  La \textbf{dominancia digital} del candidato $c$ es:
  \[
    \delta_c^{(d)} = \frac{\sum_{p \in P_c} \text{likes}_p}
                          {\displaystyle\sum_{c' \in \mathcal{C}_d}\sum_{p \in P_{c'}} \text{likes}_p}
  \]
  donde $P_c$ son las publicaciones del candidato $c$ en la ventana de 14 días previos.\\[0.6em]
  \textbf{Propiedad clave}: $\displaystyle\sum_{c \in \mathcal{C}_d} \delta_c^{(d)} = 1$
  \quad {\color{elanGray}\small(suma cero dentro de contienda)}
