\begin{frame}{Limitación estructural identificada}
\vspace{0.5em}
\begin{columns}[c]
\column{0.52\textwidth}
  \begin{tikzpicture}
    \fill[elanRed!15!elanPanel] (0,0) rectangle (6.5,4.5);
    \draw[elanRed, line width=1pt] (0,0) rectangle (6.5,4.5);
    \node[anchor=north, yshift=-0.3cm, text=elanRed, font=\bfseries\small] at (3.25,4.5)
      {Lo que SOMEN-DC NO puede hacer};
    \node[anchor=north, yshift=-1.0cm, text width=6cm] at (3.25,4.5) {
      \begin{itemize}\itemsep0.5em
        \item[\color{elanRed}$\times$] \small Predecir elecciones \textbf{sin encuestas históricas} del mismo país
        \item[\color{elanRed}$\times$] \small Transferirse a elecciones de \textbf{formato diferente} (local $\to$ presidencial)
        \item[\color{elanRed}$\times$] \small Operar sobre datos digitales \textbf{reconstruidos históricamente}
      \end{itemize}
    };
  \end{tikzpicture}

\column{0.44\textwidth}
  \begin{tikzpicture}
    \fill[elanGreen!15!elanPanel] (0,0) rectangle (5.5,4.5);
    \draw[elanGreen, line width=1pt] (0,0) rectangle (5.5,4.5);
    \node[anchor=north, yshift=-0.3cm, text=elanGreen, font=\bfseries\small] at (2.75,4.5)
      {Lo que ELA-NOM resuelve};
    \node[anchor=north, yshift=-1.0cm, text width=5cm] at (2.75,4.5) {
      \begin{itemize}\itemsep0.5em
        \item[\color{elanGreen}$\checkmark$] \small Entrena exclusivamente en \textbf{datos digitales}
        \item[\color{elanGreen}$\checkmark$] \small Usa \textbf{dominancia relativa} transferible entre elecciones
        \item[\color{elanGreen}$\checkmark$] \small Reconstrucción temporal \textbf{Weibull}
      \end{itemize}
    };
  \end{tikzpicture}
\end{columns}
\vspace{0.6em}
{\color{elanGray}\small $\Rightarrow$ La Etapa 2 ataca directamente el problema de la reconstrucción temporal.}
